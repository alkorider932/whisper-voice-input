WHISPER MLX SYSTEM

Локальный голосовой ввод на Mac

Полное руководство по настройке фонового сервиса: двойной клик Ctrl запускает микрофон, нейросеть расшифровывает речь и мгновенно печатает текст в активное поле.

⚡ Двойной Ctrl

Мгновенный старт и остановка записи без переключения окон.

🎙️ Whisper Turbo

Локальная расшифровка русской речи на Neural Engine без интернета.

✍️ Автопечать

Автоматическая вставка в Telegram, Notion, браузер и любые поля.

Файл voice_daemon.py

Скрипт фонового демона:

voice_daemon.py
import os
import sys
import time
import queue
import tempfile
import threading
import subprocess
import numpy as np
import sounddevice as sd
import soundfile as sf
from pynput import keyboard
import mlx_whisper

WHISPER_MODEL = "mlx-community/whisper-large-v3-turbo"
SAMPLE_RATE = 16000
DOUBLE_CLICK_INTERVAL = 0.4

is_recording = False
audio_q = queue.Queue()
last_ctrl_time = 0
record_thread = None

def play_sound(sound_name="Tink"):
    subprocess.Popen(["afplay", f"/System/Library/Sounds/{sound_name}.aiff"],
                     stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

def audio_callback(indata, frames, time_info, status):
    if is_recording:
        audio_q.put(indata.copy())

def type_text(text: str):
    if not text:
        return
    p = subprocess.Popen(["pbcopy"], stdin=subprocess.PIPE, text=True)
    p.communicate(input=text)
    
    applescript = """
    tell application "System Events"
        keystroke "v" using command down
    end tell
    """
    subprocess.run(["osascript", "-e", applescript])

def recording_worker():
    global is_recording
    frames = []
    
    with sd.InputStream(samplerate=SAMPLE_RATE, channels=1, dtype="float32", callback=audio_callback):
        while is_recording:
            try:
                data = audio_q.get(timeout=0.1)
                frames.append(data)
            except queue.Empty:
                pass

    if not frames:
        return

    audio_data = np.concatenate(frames, axis=0)
    
    with tempfile.NamedTemporaryFile(suffix=".wav", delete=False) as tmp:
        tmp_path = tmp.name
        sf.write(tmp_path, audio_data, SAMPLE_RATE)

    try:
        print("🎙️ Расшифровка через Whisper MLX...")
        result = mlx_whisper.transcribe(
            tmp_path,
            path_or_hf_repo=WHISPER_MODEL,
            language="ru"
        )
        text = result.get("text", "").strip()
        print(f"✍️ Распознано: {text}")
        type_text(text)
    except Exception as e:
        print(f"⚠️ Ошибка Whisper: {e}")
    finally:
        if os.path.exists(tmp_path):
            os.remove(tmp_path)

def toggle_recording():
    global is_recording, record_thread
    if not is_recording:
        is_recording = True
        while not audio_q.empty():
            audio_q.get_nowait()
        play_sound("Blow")
        print("\n🔴 Запись началась... Говорите!")
        record_thread = threading.Thread(target=recording_worker)
        record_thread.start()
    else:
        is_recording = False
        play_sound("Pop")
        print("⏹️ Запись остановлена. Обработка...")

def on_press(key):
    global last_ctrl_time
    if key in (keyboard.Key.ctrl, keyboard.Key.ctrl_l, keyboard.Key.ctrl_r):
        now = time.time()
        if now - last_ctrl_time < DOUBLE_CLICK_INTERVAL:
            last_ctrl_time = 0
            toggle_recording()
        else:
            last_ctrl_time = now

def main():
    print("="*60)
    print("🚀 Фоновый голосовой ввод активен!")
    print("👉 Нажмите Ctrl 2 раза подряд, чтобы НАЧАТЬ запись.")
    print("👉 Нажмите Ctrl 2 раза подряд, чтобы ОСТАНОВИТЬ запись и вставить текст.")
    print("="*60)
    
    with keyboard.Listener(on_press=on_press) as listener:
        listener.join()

if __name__ == "__main__":
    main()

Разрешения macOS и первый запуск

Выдайте необходимые права доступа:

  1. Откройте Настройки Mac → Конфиденциальность и безопасность.
  2. В разделах Универсальный доступ и Мониторинг ввода включите Терминал / VS Code.
  3. В разделе Микрофон разрешите доступ Терминалу.

Запуск:

Терминал macOS
./venv/bin/python3 voice_daemon.py
5

Автозапуск в фоне (LaunchAgent)

Чтобы голосовой ввод работал автоматически без открытого терминала:

com.user.voicetyper.plist
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
    <key>Label</key>
    <string>com.user.voicetyper</string>
    <key>ProgramArguments</key>
    <array>
        <string>/Users/CURRENT_USER/Desktop/voice-typer/venv/bin/python3</string>
        <string>/Users/CURRENT_USER/Desktop/voice-typer/voice_daemon.py</string>
    </array>
    <key>RunAtLoad</key>
    <true/>
    <key>KeepAlive</key>
    <true/>
</dict>
</plist>
Замените CURRENT_USER на имя вашей учётной записи в macOS и положите файл в ~/Library/LaunchAgents/.
